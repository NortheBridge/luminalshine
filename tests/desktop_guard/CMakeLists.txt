cmake_minimum_required(VERSION 3.20)
project(desktop_guard_regressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
add_subdirectory("${REPO_ROOT}/third-party/googletest" googletest)
add_executable(desktop_guard_tests
  ../unit/test_exact_device_window_router_policy.cpp
  ../unit/test_desktop_guard_win32.cpp)
target_include_directories(desktop_guard_tests PRIVATE "${REPO_ROOT}")
target_link_libraries(desktop_guard_tests PRIVATE GTest::gtest_main)
if(WIN32)
  target_compile_definitions(desktop_guard_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(desktop_guard_tests PRIVATE user32)
endif()
enable_testing()
add_test(NAME desktop_guard_regressions COMMAND desktop_guard_tests)
